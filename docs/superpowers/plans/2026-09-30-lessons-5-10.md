#!/usr/bin/env python3
"""Генерира HTML версиите на уроците от Markdown.

Пускане от папка csharp-for-beginners:

    python tools/build_html.py

Чете всички lessons/lesson-*.md и записва html/lessons/lesson-*.html.
Поддържа само Markdown елементите, които уроците използват. Ако срещне
непознат елемент, спира с грешка (файл и ред), вместо тихо да развали
страницата. HTML уроците не се редактират на ръка.
"""

from __future__ import annotations

import html
import re
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent

PAGE_HEAD = """<!doctype html>
<html lang="bg">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <meta http-equiv="Cache-Control" content="no-store, no-cache, must-revalidate, max-age=0">
  <meta http-equiv="Pragma" content="no-cache">
  <meta http-equiv="Expires" content="0">
  <title>__TITLE__</title>
  <link rel="stylesheet" href="../assets/course.css">
  <script>
    (function () {
      var version = Date.now().toString();

      function versionedUrl(href) {
        try {
          var url = new URL(href, window.location.href);
          if (url.origin !== window.location.origin) {
            return href;
          }
          url.searchParams.set("v", version);
          return url.href;
        } catch (error) {
          return href;
        }
      }

      function applyCacheBusting() {
        document.querySelectorAll("link[rel=\\"stylesheet\\"][href]").forEach(function (link) {
          link.setAttribute("href", versionedUrl(link.getAttribute("href")));
        });

        document.querySelectorAll("a[href]").forEach(function (link) {
          var href = link.getAttribute("href");
          if (/\\.html(?:$|[?#])/.test(href)) {
            link.setAttribute("href", versionedUrl(href));
          }
        });
      }

      if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", applyCacheBusting);
      } else {
        applyCacheBusting();
      }
    }());
  </script>
</head>
<body>
  <main class="page">
    <nav class="top-nav" aria-label="Основна навигация">
      <div class="top-nav-title">C# за начинаещи</div>
      <div class="top-nav-links">
        <a href="../index.html">Начало</a>
        <a href="../docs/curriculum.html">Програма</a>
        <a href="../docs/glossary.html">Речник</a>
      </div>
    </nav>

    <article class="lesson">
      <header class="lesson-header">
        <div class="lesson-label">__LABEL__</div>
        <h1>__HEADING__</h1>
      </header>"""

PAGE_FOOT = """
      <footer class="footer-nav">
        Важни страници: <a href="../index.html">Начало</a>, <a href="../docs/curriculum.html">Програма</a>, <a href="../docs/glossary.html">Речник</a>.
      </footer>
    </article>
  </main>
</body>
</html>
"""

TITLE_RE = re.compile(r"^(Урок \d+): (.+)$")
HEADING_RE = re.compile(r"^(#+) (.+)$")
FENCE_RE = re.compile(r"^```(.*)$")
UL_RE = re.compile(r"^- (.*)$")
OL_RE = re.compile(r"^(\d+)\. (.*)$")
TABLE_SEPARATOR_RE = re.compile(r"^\|(\s*:?-+:?\s*\|)+\s*$")
BOLD_RE = re.compile(r"\*\*(.+?)\*\*")
CELL_SPLIT_RE = re.compile(r"(?<!\\)\|")

CODE_CLASSES = {"csharp": ' class="language-csharp"', "text": "", "": ""}
SECTION_WRAPPERS = {"Практика": "practice-block", "Упражнения по трудност": "difficulty"}


class BuildError(Exception):
    """Markdown, който скриптът не може да преобразува."""


def render_inline(text: str) -> str:
    """Екранира текста и превръща `код` и **удебелено** в HTML."""
    parts = text.split("`")
    if len(parts) % 2 == 0:
        raise ValueError("незатворен ` в текста")
    out = []
    for index, part in enumerate(parts):
        escaped = html.escape(part, quote=False)
        if index % 2 == 1:
            out.append("<code>" + escaped + "</code>")
        else:
            out.append(BOLD_RE.sub(r"<strong>\1</strong>", escaped))
    return "".join(out)


def _starts_block(line: str) -> bool:
    return bool(
        FENCE_RE.match(line)
        or line.startswith(("#", ">", "|"))
        or UL_RE.match(line)
        or OL_RE.match(line)
    )


def _split_row(row: str) -> list[str]:
    inner = row.strip()[1:]
    if inner.endswith("|") and not inner.endswith("\\|"):
        inner = inner[:-1]
    return [cell.strip().replace("\\|", "|") for cell in CELL_SPLIT_RE.split(inner)]


def parse_blocks(md_text: str, source: str) -> list[dict]:
    """Разделя Markdown на блокове. Всеки блок пази номера на реда си."""
    lines = [line.rstrip("\r") for line in md_text.split("\n")]
    blocks: list[dict] = []
    i = 0

    def fail(line_no: int, message: str) -> None:
        raise BuildError(f"{source}:{line_no}: {message}")

    def check_indent(index: int) -> None:
        if lines[index][:1] in (" ", "\t"):
            fail(index + 1, "ред с отстъп извън код блок (вложени списъци не се поддържат)")

    while i < len(lines):
        line = lines[i]
        line_no = i + 1
        if not line.strip():
            i += 1
            continue
        check_indent(i)

        fence = FENCE_RE.match(line)
        if fence:
            lang = fence.group(1).strip()
            if lang not in CODE_CLASSES:
                fail(line_no, f"непознат език на код блок: {lang!r}")
            body = []
            i += 1
            while i < len(lines) and lines[i].strip() != "```":
                body.append(lines[i])
                i += 1
            if i == len(lines):
                fail(line_no, "незатворен код блок")
            i += 1
            blocks.append({"type": "code", "lang": lang, "lines": body, "line": line_no})
            continue

        if line.startswith("#"):
            heading = HEADING_RE.match(line)
            if not heading:
                fail(line_no, "след # трябва да има интервал")
            level = len(heading.group(1))
            if level > 3:
                fail(line_no, "заглавия с #### и повече не се поддържат")
            blocks.append({"type": f"h{level}", "text": heading.group(2).strip(), "line": line_no})
            i += 1
            continue

        if line.startswith(">"):
            parts = []
            while i < len(lines) and lines[i].startswith(">"):
                parts.append(lines[i][1:].strip())
                i += 1
            text = " ".join(part for part in parts if part)
            blocks.append({"type": "quote", "text": text, "line": line_no})
            continue

        if line.startswith("|"):
            rows = []
            while i < len(lines) and lines[i].startswith("|"):
                rows.append(lines[i])
                i += 1
            if len(rows) < 2 or not TABLE_SEPARATOR_RE.match(rows[1]):
                fail(line_no, "таблицата трябва да има ред със заглавия и ред |---|")
            header = _split_row(rows[0])
            body_rows = [_split_row(row) for row in rows[2:]]
            for offset, row in enumerate(body_rows):
                if len(row) != len(header):
                    fail(line_no + 2 + offset, "редът на таблицата има различен брой колони")
            blocks.append({"type": "table", "header": header, "rows": body_rows, "line": line_no})
            continue

        first_ul = UL_RE.match(line)
        first_ol = OL_RE.match(line)
        if first_ul or first_ol:
            item_re = UL_RE if first_ul else OL_RE
            items: list[dict] = []
            while i < len(lines):
                current = lines[i]
                item = item_re.match(current)
                if item:
                    items.append({"text": item.group(item.lastindex), "line": i + 1})
                    i += 1
                    continue
                if not current.strip():
                    next_index = i
                    while next_index < len(lines) and not lines[next_index].strip():
                        next_index += 1
                    if next_index < len(lines) and item_re.match(lines[next_index]):
                        i = next_index
                        continue
                    break
                if _starts_block(current):
                    break
                check_indent(i)
                items[-1]["text"] += " " + current.strip()
                i += 1
            block = {"type": "ul" if first_ul else "ol", "items": items, "line": line_no}
            block["start"] = int(first_ol.group(1)) if first_ol else 1
            blocks.append(block)
            continue

        parts = []
        while i < len(lines) and lines[i].strip():
            if parts and _starts_block(lines[i]):
                break
            check_indent(i)
            parts.append(lines[i].strip())
            i += 1
        blocks.append({"type": "p", "text": " ".join(parts), "line": line_no})

    return blocks


def _inline(block_text: str, source: str, line_no: int) -> str:
    try:
        return render_inline(block_text)
    except ValueError as error:
        raise BuildError(f"{source}:{line_no}: {error}") from None


def _render_block(block: dict, pad: str, source: str) -> list[str]:
    kind = block["type"]
    line_no = block["line"]
    if kind == "p":
        return [f"{pad}<p>{_inline(block['text'], source, line_no)}</p>"]
    if kind == "quote":
        return [f"{pad}<blockquote>{_inline(block['text'], source, line_no)}</blockquote>"]
    if kind == "code":
        code = "\n".join(html.escape(line, quote=False) for line in block["lines"])
        return [f"{pad}<pre><code{CODE_CLASSES[block['lang']]}>{code}</code></pre>"]
    if kind in ("ul", "ol"):
        opening = f"<{kind}>"
        if kind == "ol" and block["start"] != 1:
            opening = f'<ol start="{block["start"]}">'
        out = [pad + opening]
        for item in block["items"]:
            out.append(f"{pad}  <li>{_inline(item['text'], source, item['line'])}</li>")
        out.append(f"{pad}</{kind}>")
        return out
    if kind == "table":
        out = [f"{pad}<table>", f"{pad}  <thead>"]
        cells = "".join(f"<th>{_inline(cell, source, line_no)}</th>" for cell in block["header"])
        out += [f"{pad}    <tr>{cells}</tr>", f"{pad}  </thead>", f"{pad}  <tbody>"]
        for row in block["rows"]:
            cells = "".join(f"<td>{_inline(cell, source, line_no)}</td>" for cell in row)
            out.append(f"{pad}    <tr>{cells}</tr>")
        out += [f"{pad}  </tbody>", f"{pad}</table>"]
        return out
    raise BuildError(f"{source}:{line_no}: неочакван блок {kind}")


def render_lesson(md_text: str, source: str) -> str:
    """Превръща един Markdown урок в цяла HTML страница."""
    blocks = parse_blocks(md_text, source)
    if not blocks or blocks[0]["type"] != "h1":
        raise BuildError(f"{source}:1: урокът трябва да започва с '# Урок N: Заглавие'")
    title = TITLE_RE.match(blocks[0]["text"])
    if not title:
        raise BuildError(f"{source}:{blocks[0]['line']}: заглавието трябва да е '# Урок N: Заглавие'")
    rest = blocks[1:]
    if rest and rest[0]["type"] != "h2":
        raise BuildError(f"{source}:{rest[0]['line']}: текст преди първото ## заглавие")

    page_title = html.escape(blocks[0]["text"].replace("`", ""), quote=False)
    heading = _inline(title.group(2), source, blocks[0]["line"])
    out = [
        PAGE_HEAD.replace("__TITLE__", page_title)
        .replace("__LABEL__", title.group(1))
        .replace("__HEADING__", heading)
    ]

    wrapper = None
    wrapper_open = False
    section_open = False
    for block in rest:
        kind = block["type"]
        if kind == "h1":
            raise BuildError(f"{source}:{block['line']}: второ # заглавие")
        if kind == "h2":
            if wrapper_open:
                out.append("        </div>")
                wrapper_open = False
            if section_open:
                out.append("      </section>")
            out += ["", "      <section>"]
            out.append(f"        <h2>{_inline(block['text'], source, block['line'])}</h2>")
            section_open = True
            wrapper = SECTION_WRAPPERS.get(block["text"])
            continue
        if kind == "h3":
            if wrapper_open:
                out.append("        </div>")
            if not out[-1].lstrip().startswith("<h2>"):
                out.append("")
            h3 = f"<h3>{_inline(block['text'], source, block['line'])}</h3>"
            if wrapper:
                out.append(f'        <div class="{wrapper}">')
                out.append("          " + h3)
                wrapper_open = True
            else:
                out.append("        " + h3)
            continue
        pad = "          " if wrapper_open else "        "
        out += _render_block(block, pad, source)

    if wrapper_open:
        out.append("        </div>")
    if section_open:
        out.append("      </section>")
    return "\n".join(out) + "\n" + PAGE_FOOT


def build_all(root: Path = ROOT) -> list[Path]:
    """Генерира HTML за всички уроци и връща пътищата на записаните файлове."""
    written = []
    for md_path in sorted((root / "lessons").glob("lesson-*.md")):
        page = render_lesson(md_path.read_text(encoding="utf-8"), md_path.name)
        out_path = root / "html" / "lessons" / (md_path.stem + ".html")
        out_path.write_text(page, encoding="utf-8", newline="\n")
        written.append(out_path)
    return written


def main() -> int:
    for stream in (sys.stdout, sys.stderr):
        if hasattr(stream, "reconfigure"):
            stream.reconfigure(encoding="utf-8", errors="replace")
    try:
        written = build_all()
    except BuildError as error:
        print(f"Грешка: {error}", file=sys.stderr)
        return 1
    for path in written:
        print("записан " + path.relative_to(ROOT).as_posix())
    return 0


if __name__ == "__main__":
    sys.exit(main())

```python
"""Проверки на истинските уроци и HTML страници.

Пускане от папка csharp-for-beginners:

    python -m unittest discover -s tools -v

Проверява, че HTML уроците са генерирани от текущия Markdown, съдържат целия
му текст, имат балансирани тагове и работещи линкове, че всеки урок има линк
от началната страница и програмата, и че речникът в Markdown и HTML съвпада.
"""

from __future__ import annotations

import html
import re
import sys
import unittest
from html.parser import HTMLParser
from pathlib import Path

TOOLS = Path(__file__).resolve().parent
sys.path.insert(0, str(TOOLS))

import build_html  # noqa: E402

ROOT = TOOLS.parent
VOID_TAGS = {"meta", "link", "br", "img", "hr", "input"}
SEPARATOR_RE = re.compile(r"^\|(\s*:?-+:?\s*\|)+$")
CELL_SPLIT_RE = re.compile(r"(?<!\\)\|")


def lesson_paths() -> list[Path]:
    return sorted((ROOT / "lessons").glob("lesson-*.md"))


def visible_text(page: str) -> str:
    """Текстът в <article>, както го вижда читателят."""
    body = page.split('<article class="lesson">', 1)[1].split("</article>", 1)[0]
    body = re.sub(r"</?(code|strong)>", "", body)
    body = re.sub(r"<[^>]+>", " ", body)
    return " ".join(html.unescape(body).split())


def table_cells(line: str) -> list[str]:
    return [cell.strip().replace("\\|", "|") for cell in CELL_SPLIT_RE.split(line.strip().strip("|"))]


def markdown_lines(md_text: str) -> list[str]:
    """Редовете на Markdown урока без Markdown знаците, така както трябва да се виждат в HTML."""
    result = []
    for raw in md_text.split("\n"):
        line = raw.strip()
        if not line or line.startswith("```") or SEPARATOR_RE.match(line):
            continue
        if line.startswith("# "):
            label, _, heading = line[2:].partition(": ")
            result += [label, heading.replace("`", "")]
            continue
        if line.startswith("|"):
            result += [cell.replace("`", "") for cell in table_cells(line) if cell]
            continue
        line = re.sub(r"^(#{2,3} |> ?|- |\d+\. )", "", line)
        result.append(" ".join(line.replace("`", "").replace("**", "").split()))
    return [line for line in result if line]


class TagBalanceChecker(HTMLParser):
    def __init__(self) -> None:
        super().__init__()
        self.stack: list[str] = []
        self.problems: list[str] = []

    def handle_starttag(self, tag: str, attrs) -> None:
        if tag not in VOID_TAGS:
            self.stack.append(tag)

    def handle_endtag(self, tag: str) -> None:
        if not self.stack or self.stack[-1] != tag:
            self.problems.append(f"</{tag}> на ред {self.getpos()[0]}, отворен е {self.stack[-1:]}")
        else:
            self.stack.pop()


class SiteTests(unittest.TestCase):
    def test_there_are_lessons(self):
        self.assertTrue(lesson_paths(), "няма уроци в lessons/")

    def test_generated_html_is_up_to_date(self):
        for md_path in lesson_paths():
            html_path = ROOT / "html" / "lessons" / (md_path.stem + ".html")
            with self.subTest(lesson=md_path.name):
                self.assertTrue(html_path.exists(), f"липсва {html_path.name}")
                expected = build_html.render_lesson(md_path.read_text(encoding="utf-8"), md_path.name)
                self.assertEqual(html_path.read_text(encoding="utf-8"), expected,
                                 "HTML не е генериран от текущия Markdown; пусни python tools/build_html.py")

    def test_html_contains_all_markdown_text(self):
        for md_path in lesson_paths():
            md_text = md_path.read_text(encoding="utf-8")
            text = visible_text(build_html.render_lesson(md_text, md_path.name))
            missing = [line for line in markdown_lines(md_text) if line not in text]
            with self.subTest(lesson=md_path.name):
                self.assertEqual(missing, [])

    def test_all_html_pages_have_balanced_tags(self):
        for page_path in sorted((ROOT / "html").rglob("*.html")):
            checker = TagBalanceChecker()
            checker.feed(page_path.read_text(encoding="utf-8"))
            checker.close()
            with self.subTest(page=page_path.relative_to(ROOT).as_posix()):
                self.assertEqual(checker.problems, [])
                self.assertEqual(checker.stack, [])

    def test_all_relative_links_point_to_existing_files(self):
        for page_path in sorted((ROOT / "html").rglob("*.html")):
            page = page_path.read_text(encoding="utf-8")
            for href in re.findall(r'href="([^"]+)"', page):
                if re.match(r"^(https?:|mailto:|#)", href):
                    continue
                target = (page_path.parent / href.split("#")[0].split("?")[0]).resolve()
                with self.subTest(page=page_path.name, href=href):
                    self.assertTrue(target.exists(), f"{href} не съществува")

    def test_every_lesson_is_linked_from_index_and_curriculum(self):
        index = (ROOT / "html" / "index.html").read_text(encoding="utf-8")
        curriculum = (ROOT / "html" / "docs" / "curriculum.html").read_text(encoding="utf-8")
        for md_path in lesson_paths():
            name = md_path.stem + ".html"
            with self.subTest(lesson=name):
                self.assertIn(f'href="lessons/{name}"', index)
                self.assertIn(f'href="../lessons/{name}"', curriculum)

    def test_glossary_markdown_matches_html(self):
        md_rows = []
        for line in (ROOT / "docs" / "glossary.md").read_text(encoding="utf-8").splitlines():
            line = line.strip()
            if line.startswith("|") and not SEPARATOR_RE.match(line):
                md_rows.append(table_cells(line))
        page = (ROOT / "html" / "docs" / "glossary.html").read_text(encoding="utf-8")
        html_rows = []
        for row in re.findall(r"<tr>(.*?)</tr>", page, re.S):
            cells = re.findall(r"<t[hd]>(.*?)</t[hd]>", row, re.S)
            html_rows.append([html.unescape(re.sub(r"</?code>", "`", cell)).strip() for cell in cells])
        self.assertEqual(md_rows, html_rows)


if __name__ == "__main__":
    unittest.main()

```python
#!/usr/bin/env python3
"""Проверява C# примерите в уроците. Временен инструмент, не е част от repo-то.

Употреба:
    python check_examples.py <урок.md> [<урок.md> ...]

Всеки ```csharp блок става отделен статичен метод в общ .NET 10 проект.
Компилира се на кръгове (компилаторът спира след синтактичните грешки), докато
останат само блоковете без грешки. Те се пускат с входа от урока и изходът се
сравнява с изхода, показан в урока.

Изпълнения на един блок се взимат от text блоковете след него (в същата подсекция):
  "Ако човекът въведе:" / "Ако човекът напише:" / "Потребителят въвежда:" -> вход
  "Резултат:" / "...резултатът ще бъде:" / "...ще видиш:"                   -> точен изход
  "Примерно изпълнение:"                   -> разговор (изход + въведеното, в реда на конзолата)
Нов вход или втори резултат започват ново изпълнение. "Потребителят въвежда:" може
да е и точно преди кода.

Категории:
  working  - трябва да се компилира и изходът да съвпада
  broken   - нарочно грешен пример ("Чести грешки" преди "Поправка:", "Намери бъга",
             "Поправи кода", или текст преди кода с "поправи", "грешк", "бъг", "не е правил");
             резултатът се проверява на ръка спрямо описанието в урока
  fragment - парче без ; и без { (напр. `7 / 2`); не се компилира

manifest.json до скрипта уточнява блок по номера му в урока (#1, #2, ...):
    {"lesson-05.md": {"7": {"stdin": ["15"], "category": "fragment", "skip_compare": true}}}
"stdin" се отнася за първото изпълнение.

Изходен код 1, ако има проблем при "working" блок.
"""

from __future__ import annotations

import json
import re
import subprocess
import sys
from pathlib import Path

HERE = Path(__file__).resolve().parent
PROJECT = HERE / "proj"
MANIFEST = HERE / "manifest.json"
CODE_OFFSET = 5  # първият ред на блока е 5-ти ред в генерирания .cs файл

CSPROJ = """<Project Sdk="Microsoft.NET.Sdk">
  <PropertyGroup>
    <OutputType>Exe</OutputType>
    <TargetFramework>net10.0</TargetFramework>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>enable</Nullable>
    <InvariantGlobalization>true</InvariantGlobalization>
    <NoWarn>CS8600;CS8602;CS8604;CS0219;CS0168</NoWarn>
  </PropertyGroup>
</Project>
"""

PROGRAM = """using System.Globalization;
using System.Reflection;
using System.Text;

CultureInfo.DefaultThreadCurrentCulture = CultureInfo.InvariantCulture;
CultureInfo.CurrentCulture = CultureInfo.InvariantCulture;
var utf8 = new UTF8Encoding(false);
Console.SetIn(new StreamReader(Console.OpenStandardInput(), utf8));
Console.SetOut(new StreamWriter(Console.OpenStandardOutput(), utf8) { AutoFlush = true });
try
{
    Type.GetType(args[0])!.GetMethod("Run")!.Invoke(null, null);
}
catch (TargetInvocationException e)
{
    Console.WriteLine("!!! " + e.InnerException!.GetType().Name + ": " + e.InnerException.Message);
    return 3;
}
return 0;
"""

HEADING_RE = re.compile(r"^(#{2,3}) (.+)$")
DIAG_RE = re.compile(r"(B\d+_\d+)\.cs\((\d+),\d+\): (error|warning) (CS\d+): (.*?)(?: \[[^\]]*\])?$")
STDIN_LEADS = ("ако човекът въведе:", "ако потребителят въведе:", "потребителят въвежда:",
               "ако човекът напише:", "ако потребителят напише:")
STDOUT_LEADS = ("резултат:", "изход:")
STDOUT_ENDINGS = ("резултатът ще бъде:", "резултатът е:", "ще видиш:")
TRANSCRIPT_LEADS = ("примерно изпълнение:",)
BUG_WORDS = re.compile(r"поправи|грешк|бъг|не е правил", re.IGNORECASE)
PARTIAL_CODES = {"CS0103", "CS0841"}  # кратък ред, който ползва променливи от предишен пример
BAD = ("COMPILE-ERROR", "TIMEOUT", "NEEDS-INPUT", "EXCEPTION", "MISMATCH", "MISMATCH-TRANSCRIPT")


def extract_blocks(md_path: Path) -> list[dict]:
    """Всички код блокове на урока с контекст: заглавия, ред, текстът точно преди блока."""
    lines = [line.rstrip("\r") for line in md_path.read_text(encoding="utf-8").split("\n")]
    blocks: list[dict] = []
    h2 = h3 = lead = ""
    after_fix = False
    i = 0
    while i < len(lines):
        line = lines[i]
        heading = HEADING_RE.match(line)
        if heading:
            if heading.group(1) == "##":
                h2, h3 = heading.group(2), ""
            else:
                h3 = heading.group(2)
            after_fix = False
            lead = ""
            i += 1
            continue
        if line.startswith("```"):
            lang = line[3:].strip()
            first_line = i + 2
            body = []
            i += 1
            while i < len(lines) and lines[i].strip() != "```":
                body.append(lines[i])
                i += 1
            i += 1
            blocks.append({"lang": lang, "code": "\n".join(body), "line": first_line,
                           "h2": h2, "h3": h3, "lead": lead, "after_fix": after_fix})
            lead = ""
            continue
        if line.strip():
            lead = line.strip()
            if lead == "Поправка:":
                after_fix = True
        i += 1
    return blocks


def classify(block: dict) -> str:
    code = block["code"]
    if ";" not in code and "{" not in code:
        return "fragment"
    if block["h2"] == "Чести грешки" and not block["after_fix"]:
        return "broken"
    if block["h2"] == "Практика" and block["h3"] in ("Намери бъга", "Поправи кода"):
        return "broken"
    if BUG_WORDS.search(block["lead"]):
        return "broken"
    return "working"


def text_lines(text: str) -> list[str]:
    lines = [line.rstrip() for line in text.replace("\r\n", "\n").split("\n")]
    while lines and not lines[-1]:
        lines.pop()
    return lines


def find_runs(blocks: list[dict], position: int) -> list[dict]:
    """Изпълненията (вход и очакван изход), описани около код блока."""
    block = blocks[position]
    same_place = lambda other: other["h2"] == block["h2"] and other["h3"] == block["h3"]
    runs: list[dict] = []
    current = {"stdin": None, "expected": None, "transcript": None}
    if position > 0:
        previous = blocks[position - 1]
        if previous["lang"] != "csharp" and same_place(previous) \
                and previous["lead"].lower() in STDIN_LEADS:
            current["stdin"] = text_lines(previous["code"])
    for later in blocks[position + 1:]:
        if later["lang"] == "csharp" or not same_place(later):
            break
        lead = later["lead"].lower()
        if lead in STDIN_LEADS:
            kind = "stdin"
        elif lead in STDOUT_LEADS or lead.endswith(STDOUT_ENDINGS):
            kind = "expected"
        elif lead in TRANSCRIPT_LEADS:
            kind = "transcript"
        else:
            continue
        starts_new = current[kind] is not None or (
            kind == "stdin" and (current["expected"] is not None or current["transcript"] is not None))
        if starts_new:
            runs.append(current)
            current = {"stdin": None, "expected": None, "transcript": None}
        current[kind] = text_lines(later["code"])
    runs.append(current)
    return runs


def collect_cases(md_paths: list[Path], manifest: dict) -> list[dict]:
    cases = []
    for md_path in md_paths:
        lesson_no = int(re.search(r"(\d+)", md_path.stem).group(1))
        blocks = extract_blocks(md_path)
        overrides = manifest.get(md_path.name, {})
        index = 0
        for position, block in enumerate(blocks):
            if block["lang"] != "csharp":
                continue
            index += 1
            case = {"file": md_path.name, "index": index, "line": block["line"],
                    "where": f"{block['h2']} / {block['h3'] or '-'}",
                    "code": block["code"], "category": classify(block),
                    "cls": f"B{lesson_no:02d}_{index:03d}",
                    "runs": find_runs(blocks, position), "skip_compare": False,
                    "errors": [], "warnings": []}
            override = overrides.get(str(index), {})
            if "stdin" in override:
                case["runs"][0]["stdin"] = override["stdin"]
            for key in ("category", "skip_compare"):
                if key in override:
                    case[key] = override[key]
            cases.append(case)
    return cases


def write_project(cases: list[dict]) -> None:
    PROJECT.mkdir(parents=True, exist_ok=True)
    for old in PROJECT.glob("B*.cs"):
        old.unlink()
    (PROJECT / "check.csproj").write_text(CSPROJ, encoding="utf-8")
    (PROJECT / "Program.cs").write_text(PROGRAM, encoding="utf-8")
    for case in cases:
        source = (f"static class {case['cls']}\n{{\n    public static void Run()\n    {{\n"
                  f"{case['code']}\n    }}\n}}\n")
        (PROJECT / f"{case['cls']}.cs").write_text(source, encoding="utf-8")


def build() -> tuple[bool, dict]:
    result = subprocess.run(
        ["dotnet", "build", str(PROJECT / "check.csproj"), "-nologo", "-v", "q", "-clp:NoSummary"],
        capture_output=True, text=True, encoding="utf-8", errors="replace")
    diagnostics: dict[str, set] = {}
    for line in result.stdout.splitlines():
        match = DIAG_RE.search(line)
        if match:
            cls, file_line, severity, code, message = match.groups()
            diagnostics.setdefault(cls, set()).add((severity, code, int(file_line), message.strip()))
    return result.returncode == 0, diagnostics


def compile_all(cases: list[dict]) -> bool:
    """Компилира на кръгове и попълва errors/warnings на всеки блок."""
    warnings: dict[str, set] = {}
    runnable = [case for case in cases if case["category"] != "fragment"]
    for _ in range(8):
        write_project(runnable)
        ok, diagnostics = build()
        for cls, found in diagnostics.items():
            warnings.setdefault(cls, set()).update(d for d in found if d[0] == "warning")
        if ok:
            break
        failed = False
        for case in runnable:
            errors = sorted(d for d in diagnostics.get(case["cls"], set()) if d[0] == "error")
            if errors:
                case["errors"] = errors
                failed = True
        if not failed:
            print("Компилирането се провали без грешка в конкретен блок:", diagnostics)
            return False
        runnable = [case for case in runnable if not case["errors"]]
    else:
        print("Компилирането не успя след 8 кръга.")
        return False
    for case in cases:
        case["warnings"] = sorted(warnings.get(case["cls"], set()))
    return True


def execute(cls: str, stdin: list[str] | None) -> tuple[str, list[str]]:
    dll = PROJECT / "bin" / "Debug" / "net10.0" / "check.dll"
    data = "".join(line + "\n" for line in (stdin or []))
    try:
        result = subprocess.run(["dotnet", str(dll), cls], input=data.encode("utf-8"),
                                capture_output=True, timeout=10)
    except subprocess.TimeoutExpired:
        return "TIMEOUT", []
    output = text_lines(result.stdout.decode("utf-8", errors="replace"))
    return ("EXCEPTION" if result.returncode == 3 else "RAN"), output


def remove_inputs(transcript: list[str], inputs: list[str]) -> list[str]:
    remaining = list(transcript)
    start = 0
    for value in inputs:
        for position in range(start, len(remaining)):
            if remaining[position] == value:
                del remaining[position]
                start = position
                break
    return remaining


def evaluate(case: dict) -> tuple[str, list[str]]:
    details = []
    for severity, code, file_line, message in case["errors"] + case["warnings"]:
        details.append(f"{severity} {code} (ред {case['line'] + file_line - CODE_OFFSET}): {message}")
    if case["errors"]:
        short = len(case["code"].strip().splitlines()) <= 3
        if short and case["category"] == "working" and {d[1] for d in case["errors"]} <= PARTIAL_CODES:
            return "PARTIAL", details
        return "COMPILE-ERROR", details

    statuses = []
    for number, run in enumerate(case["runs"], start=1):
        status, output = execute(case["cls"], run["stdin"])
        prefix = f"изпълнение {number}: " if len(case["runs"]) > 1 else ""
        if status == "EXCEPTION" and run["stdin"] is None and "Console.ReadLine" in case["code"]:
            status = "NEEDS-INPUT"
        elif status == "RAN" and not case["skip_compare"]:
            if run["expected"] is not None:
                status = "MATCH" if output == run["expected"] else "MISMATCH"
                if status == "MISMATCH":
                    details.append(prefix + "очаквано: " + " | ".join(run["expected"]))
            elif run["transcript"] is not None:
                produced = remove_inputs(run["transcript"], run["stdin"] or [])
                status = "MATCH" if output == produced else "MISMATCH-TRANSCRIPT"
                if status == "MISMATCH-TRANSCRIPT":
                    details.append(prefix + "очаквано: " + " | ".join(produced))
        if run["stdin"]:
            details.append(prefix + "вход: " + " | ".join(run["stdin"]))
        details.append(prefix + "изход: " + " | ".join(output[:14]) + (" ..." if len(output) > 14 else ""))
        statuses.append(status)
    for bad in BAD:
        if bad in statuses:
            return bad, details
    return ("MATCH" if "MATCH" in statuses else "RAN"), details


def main(argv: list[str]) -> int:
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    manifest = json.loads(MANIFEST.read_text(encoding="utf-8")) if MANIFEST.exists() else {}
    cases = collect_cases([Path(arg) for arg in argv], manifest)
    if not compile_all(cases):
        return 2

    problems = 0
    for case in cases:
        location = f"{case['file']}:{case['line']} #{case['index']}"
        header = f"{location:<24} {case['category']:<8} [{case['where']}]"
        if case["category"] == "fragment":
            print(f"{header}  FRAGMENT")
            continue
        status, details = evaluate(case)
        bad = case["category"] == "working" and status in BAD
        problems += bad
        print(f"{header}  {status}{'  <-- ПРОБЛЕМ' if bad else ''}")
        for detail in details:
            print(f"      {detail}")
    print(f"\nБлокове: {len(cases)}, проблеми при working: {problems}")
    return 1 if problems else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

Създай `$TMP/ivan-check/check_concepts.py`:

```python
#!/usr/bin/env python3
"""Търси в C# кода на урок понятия, въведени в по-късен урок (spec, раздел 4.5).

Употреба:
    python check_concepts.py <урок.md> [<урок.md> ...]

Проверява само ```csharp блоковете. Номерът на урока се взима от името на файла.
Изходен код 1, ако има нарушение.
"""

from __future__ import annotations

import re
import sys
from pathlib import Path

NEVER = 99  # не се използва в уроци 5-10

# (регулярен израз, урок на въвеждане, описание, търси ли се в суровия код)
RULES = [
    (r'\$"', NEVER, 'интерполация $"..."', True),
    (r"\\t", 10, "табулация \\t", True),
    (r"//", 5, "коментар //", True),
    (r"\bstring\.Format\b", NEVER, "string.Format", False),
    (r"\bMath\.", NEVER, "Math.*", False),
    (r"\bTryParse\b", NEVER, "TryParse", False),
    (r"\bout\b", NEVER, "out", False),
    (r"\bvar\b", NEVER, "var", False),
    (r"\bcontinue\s*;", NEVER, "continue", False),
    (r"\bcase\s*[<>!]", NEVER, "шаблон в case", False),
    (r"\w\s*\[|\[\s*\]", NEVER, "масив", False),
    (r"=>", NEVER, "=> (switch израз или ламбда)", False),
    (r"\?", NEVER, "? (тернарен оператор или null)", False),
    (r"%|\+=|-=|\*=|/=", 5, "% или +=, -=, *=, /=", False),
    (r"\bbool\b|\btrue\b|\bfalse\b", 6, "bool, true, false", False),
    (r"==|!=|<=|>=|<|>", 6, "сравнение", False),
    (r"\bif\b|\belse\b", 6, "if/else", False),
    (r"&&|\|\||!(?!=)", 6, "&&, ||, !", False),
    (r"\bswitch\b|\bcase\b|\bdefault\b|\bbreak\b", 7, "switch/case/break", False),
    (r"\bwhile\b|\bdo\b", 8, "while/do while", False),
    (r"\+\+|--", 8, "++ или --", False),
    (r"\bRandom\b", 8, "Random", False),
    (r"\bfor\b", 9, "for", False),
    (r"Console\.Write\s*\(", 9, "Console.Write", False),
    (r"Console\.WriteLine\s*\(\s*\)", 9, "Console.WriteLine() без аргументи", False),
]

STRING_RE = re.compile(r'"(?:[^"\\\n]|\\.)*"')
COMMENT_RE = re.compile(r"//[^\n]*")


def csharp_blocks(md_path: Path) -> list[tuple[int, str]]:
    lines = md_path.read_text(encoding="utf-8").split("\n")
    blocks, i = [], 0
    while i < len(lines):
        if lines[i].strip() == "```csharp":
            start, body = i + 2, []
            i += 1
            while i < len(lines) and lines[i].strip() != "```":
                body.append(lines[i])
                i += 1
            blocks.append((start, "\n".join(body)))
        i += 1
    return blocks


def main(argv: list[str]) -> int:
    sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    violations = 0
    for arg in argv:
        md_path = Path(arg)
        lesson_no = int(re.search(r"(\d+)", md_path.stem).group(1))
        for start, code in csharp_blocks(md_path):
            cleaned = COMMENT_RE.sub("", STRING_RE.sub('""', code))
            for pattern, introduced, label, raw in RULES:
                if introduced <= lesson_no:
                    continue
                target = code if raw else cleaned
                for match in re.finditer(pattern, target):
                    line_no = start + target[:match.start()].count("\n")
                    when = ("не се използва в уроци 5-10" if introduced == NEVER
                            else f"въвежда се в урок {introduced}")
                    print(f"{md_path.name}:{line_no}: {label} ({when})")
                    violations += 1
                    break
    print(f"Нарушения: {violations}")
    return 1 if violations else 0


if __name__ == "__main__":
    sys.exit(main(sys.argv[1:]))

**Как мисли програмистът:** изброй всички възможни стойности; всяка е `case`; какво става при неочаквана стойност (`default`); точни стойности ли са (тогава `switch`), или интервали (тогава `if`).

**Какво ще научиш:**

{
    sum += number;
    Console.WriteLine("Въведи число (0 за край):");
    text = Console.ReadLine();
    number = int.Parse(text);
}

```

„Преди да пишеш код, планирай:“: коя променлива пази избора; кои въпроси за всяка фигура; колко символа на ред за всяка фигура; кога програмата спира. Еталонно решение (не влиза в урока; проверено и с вход `3`, `4`, `2`, `2`, `5`, `0`):

```csharp
int choice = 0;
int size = 0;
int width = 0;
string text = "";

do
{
    Console.WriteLine("ASCII художник");
    Console.WriteLine("1 - квадрат");
    Console.WriteLine("2 - правоъгълник");
    Console.WriteLine("3 - триъгълник");
    Console.WriteLine("0 - изход");
    Console.WriteLine("Избор:");
    text = Console.ReadLine();
    choice = int.Parse(text);

    switch (choice)
    {
        case 1:
            Console.WriteLine("Размер:");
            text = Console.ReadLine();
            size = int.Parse(text);
            for (int row = 1; row <= size; row++)
            {
                for (int col = 1; col <= size; col++)
                {
                    Console.Write("*");
                }
                Console.WriteLine();
            }
            break;
        case 2:
            Console.WriteLine("Височина:");
            text = Console.ReadLine();
            size = int.Parse(text);
            Console.WriteLine("Ширина:");
            text = Console.ReadLine();
            width = int.Parse(text);
            for (int row = 1; row <= size; row++)
            {
                for (int col = 1; col <= width; col++)
                {
                    Console.Write("*");
                }
                Console.WriteLine();
            }
            break;
        case 3:
            Console.WriteLine("Размер:");
            text = Console.ReadLine();
            size = int.Parse(text);
            for (int row = 1; row <= size; row++)
            {
                for (int col = 1; col <= row; col++)
                {
                    Console.Write("*");
                }
                Console.WriteLine();
            }
            break;
        case 0:
            Console.WriteLine("Довиждане!");
            break;
        default:
            Console.WriteLine("Няма такава фигура.");
            break;
    }
}
while (choice != 0);
```
